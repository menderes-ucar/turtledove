class SubjectAnalytics {
  final String subjectId;
  final String subjectName;
  final int totalAttempts;
  final int correctAnswers;
  final int wrongAnswers;
  final int blankAnswers;
  final double averageScore;
  final double net;

  SubjectAnalytics({
    required this.subjectId,
    required this.subjectName,
    required this.totalAttempts,
    required this.correctAnswers,
    required this.wrongAnswers,
    required this.blankAnswers,
    required this.averageScore,
    required this.net,
  });

  double get successPercentage {
    if (totalAttempts == 0) return 0.0;
    return (correctAnswers / totalAttempts) * 100;
  }

  factory SubjectAnalytics.fromJson(Map<String, dynamic> json) {
    return SubjectAnalytics(
      subjectId: json['subject_id'] as String? ?? '',
      subjectName: json['subject_name'] as String? ?? '',
      totalAttempts: json['total_attempts'] as int? ?? 0,
      correctAnswers: json['correct_answers'] as int? ?? 0,
      wrongAnswers: json['wrong_answers'] as int? ?? 0,
      blankAnswers: json['blank_answers'] as int? ?? 0,
      averageScore: _parseDouble(json['average_score']) ?? 0.0,
      net: _parseDouble(json['net']) ?? 0.0,
    );
  }
}

class OverallAnalytics {
  final int totalQuestionsAttempted;
  final int totalCorrect;
  final int totalWrong;
  final int totalBlank;
  final double overallSuccessPercentage;
  final double netScore;
  final List<SubjectAnalytics> subjectAnalytics;
  final int totalTestsCompleted;
  final double averageTestScore;

  OverallAnalytics({
    required this.totalQuestionsAttempted,
    required this.totalCorrect,
    required this.totalWrong,
    required this.totalBlank,
    required this.overallSuccessPercentage,
    required this.netScore,
    required this.subjectAnalytics,
    required this.totalTestsCompleted,
    required this.averageTestScore,
  });

  factory OverallAnalytics.empty() {
    return OverallAnalytics(
      totalQuestionsAttempted: 0,
      totalCorrect: 0,
      totalWrong: 0,
      totalBlank: 0,
      overallSuccessPercentage: 0.0,
      netScore: 0.0,
      subjectAnalytics: [],
      totalTestsCompleted: 0,
      averageTestScore: 0.0,
    );
  }
}

class ExamDateInfo {
  final String examName;
  final DateTime examDate;
  final int daysRemaining;
  final int targetScore;
  final double currentAverage;

  ExamDateInfo({
    required this.examName,
    required this.examDate,
    required this.targetScore,
    required this.currentAverage,
  }) : daysRemaining = examDate.difference(DateTime.now()).inDays;

  bool get isToday => daysRemaining == 0;
  bool get isTomorrow => daysRemaining == 1;
  bool get isPassed => daysRemaining < 0;
}

class AnalyticsProvider extends ChangeNotifier {
  final SupabaseClient supabaseClient;
  final String userId;

  // State
  OverallAnalytics? _overallAnalytics;
  List<SubjectAnalytics> _subjectAnalytics = [];
  ExamDateInfo? _examDateInfo;
  bool _isLoading = false;
  String? _errorMessage;

  // Getters
  OverallAnalytics? get overallAnalytics => _overallAnalytics;
  List<SubjectAnalytics> get subjectAnalytics => _subjectAnalytics;
  ExamDateInfo? get examDateInfo => _examDateInfo;
  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;

  AnalyticsProvider({
    required this.supabaseClient,
    required this.userId,
  });

  // =========================================================================
  // DATA LOADING
  // =========================================================================

  Future<void> loadAnalytics() async {
    try {
      _isLoading = true;
      _errorMessage = null;
      notifyListeners();

      // Load overall analytics
      await _loadOverallAnalytics();

      // Load subject analytics
      await _loadSubjectAnalytics();

      // Load exam date info
      await _loadExamDateInfo();

      _isLoading = false;
      notifyListeners();
    } catch (e) {
      _isLoading = false;
      _errorMessage = 'Analytics yüklenemedi: ${e.toString()}';
      notifyListeners();
      rethrow;
    }
  }

  Future<void> _loadOverallAnalytics() async {
    try {
      // Get all quiz attempts
      final data = await supabaseClient
          .from('user_quiz_attempts')
          .select()
          .eq('user_id', userId)
          .eq('status', 'completed');

      if (data.isEmpty) {
        _overallAnalytics = OverallAnalytics.empty();
        return;
      }

      int totalQuestions = 0;
      int totalCorrect = 0;
      int totalWrong = 0;
      int totalBlank = 0;
      double totalScore = 0.0;

      for (final attempt in data) {
        totalQuestions += attempt['total_questions'] as int? ?? 0;
        totalCorrect += attempt['correct_answers'] as int? ?? 0;
        totalWrong += attempt['wrong_answers'] as int? ?? 0;
        totalBlank += attempt['blank_answers'] as int? ?? 0;
        totalScore += _parseDouble(attempt['score']) ?? 0.0;
      }

      final successPercentage = totalQuestions > 0
          ? (totalCorrect / totalQuestions) * 100
          : 0.0;

      final netScore =
          totalCorrect - (totalWrong / 4);

      final averageScore = data.length > 0 ? totalScore / data.length : 0.0;

      _overallAnalytics = OverallAnalytics(
        totalQuestionsAttempted: totalQuestions,
        totalCorrect: totalCorrect,
        totalWrong: totalWrong,
        totalBlank: totalBlank,
        overallSuccessPercentage: successPercentage,
        netScore: netScore,
        subjectAnalytics: [],
        totalTestsCompleted: data.length,
        averageTestScore: averageScore,
      );
    } catch (e) {
      print('Error loading overall analytics: $e');
      _overallAnalytics = OverallAnalytics.empty();
    }
  }

  Future<void> _loadSubjectAnalytics() async {
    try {
      // Query user_answers with question details grouped by subject
      final data = await supabaseClient.rpc('get_subject_analytics', {
        'p_user_id': userId,
      });

      if (data == null || data.isEmpty) {
        _subjectAnalytics = [];
        return;
      }

      _subjectAnalytics = (data as List)
          .map((item) => SubjectAnalytics.fromJson(item as Map<String, dynamic>))
          .toList();

      // Sort by success percentage descending
      _subjectAnalytics.sort((a, b) =>
          b.successPercentage.compareTo(a.successPercentage));
    } catch (e) {
      print('Error loading subject analytics: $e');
      _subjectAnalytics = [];
    }
  }

  Future<void> _loadExamDateInfo() async {
    try {
      // Get user profile for target exam
      final profile = await supabaseClient
          .from('profiles')
          .select()
          .eq('id', userId)
          .single();

      final targetExam = profile['target_exam'] as String? ?? 'lisans';
      final targetScore = profile['target_score'] as int? ?? 70;

      // Map exam to date
      DateTime examDate;
      String examName;

      switch (targetExam) {
        case 'lisans':
          examName = 'KPSS Lisans';
          // KPSS tarih örneği - güncellenmelidir
          examDate = DateTime(2024, 5, 5);
          break;
        case 'onlisans':
          examName = 'KPSS Ön Lisans';
          examDate = DateTime(2024, 5, 5);
          break;
        case 'ortaogretim':
          examName = 'KPSS Ortaöğretim';
          examDate = DateTime(2024, 5, 5);
          break;
        default:
          examName = 'KPSS';
          examDate = DateTime(2024, 5, 5);
      }

      _examDateInfo = ExamDateInfo(
        examName: examName,
        examDate: examDate,
        targetScore: targetScore,
        currentAverage: _overallAnalytics?.averageTestScore ?? 0.0,
      );
    } catch (e) {
      print('Error loading exam date info: $e');
      _examDateInfo = null;
    }
  }

  @override
  void dispose() {
    super.dispose();
  }
}

class AnalyticsScreen extends StatefulWidget {
  const AnalyticsScreen({Key? key}) : super(key: key);

  @override
  State<AnalyticsScreen> createState() => _AnalyticsScreenState();
}

class _AnalyticsScreenState extends State<AnalyticsScreen> {
  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) {
      context.read<AnalyticsProvider>().loadAnalytics();
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.grey[50],
      appBar: AppBar(
        title: const Text('İstatistiklerim'),
        centerTitle: true,
        elevation: 0,
        backgroundColor: Colors.white,
        foregroundColor: Colors.black,
      ),
      body: Consumer<AnalyticsProvider>(
        builder: (context, analyticsProvider, _) {
          if (analyticsProvider.isLoading) {
            return const Center(child: CircularProgressIndicator());
          }

          if (analyticsProvider.errorMessage != null) {
            return Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const Icon(Icons.error_outline, size: 64, color: Colors.red),
                  const SizedBox(height: 16),
                  Text(analyticsProvider.errorMessage!),
                  const SizedBox(height: 24),
                  ElevatedButton(
                    onPressed: () =>
                        analyticsProvider.loadAnalytics(),
                    child: const Text('Yeniden Dene'),
                  ),
                ],
              ),
            );
          }

          return SingleChildScrollView(
            child: Column(
              children: [
                // Exam Countdown Banner
                if (analyticsProvider.examDateInfo != null)
                  _ExamCountdownBanner(
                    examDateInfo: analyticsProvider.examDateInfo!,
                  ),
                const SizedBox(height: 16),
                // Overall Stats
                _OverallStatsCards(
                  analytics: analyticsProvider.overallAnalytics,
                ),
                const SizedBox(height: 24),
                // Net Goal Progress
                _NetGoalProgressWidget(
                  analytics: analyticsProvider.overallAnalytics,
                  examDateInfo: analyticsProvider.examDateInfo,
                ),
                const SizedBox(height: 24),
                // Subject Analytics Bar Chart
                if (analyticsProvider.subjectAnalytics.isNotEmpty) ...[
                  _SubjectBarChart(
                    subjectAnalytics: analyticsProvider.subjectAnalytics,
                  ),
                  const SizedBox(height: 24),
                ],
                // Subject Analytics Radar Chart
                if (analyticsProvider.subjectAnalytics.isNotEmpty) ...[
                  _SubjectRadarChart(
                    subjectAnalytics: analyticsProvider.subjectAnalytics,
                  ),
                  const SizedBox(height: 24),
                ],
                // Subject Details List
                _SubjectDetailsList(
                  subjectAnalytics: analyticsProvider.subjectAnalytics,
                ),
                const SizedBox(height: 24),
              ],
            ),
          );
        },
      ),
    );
  }
}

class _NetGoalProgressWidget extends StatelessWidget {
  final OverallAnalytics? analytics;
  final ExamDateInfo? examDateInfo;

  const _NetGoalProgressWidget({
    required this.analytics,
    required this.examDateInfo,
  });

  @override
  Widget build(BuildContext context) {
    if (analytics == null || examDateInfo == null) {
      return const SizedBox.shrink();
    }

    final targetNet = (examDateInfo!.targetScore / 100) * analytics!.totalQuestionsAttempted;
    final currentNet = analytics!.netScore;
    final progress = (currentNet / targetNet).clamp(0.0, 1.0);
    final remaining = (targetNet - currentNet).clamp(0.0, double.infinity);

    return Container(
      margin: const EdgeInsets.symmetric(horizontal: 16),
      padding: const EdgeInsets.all(20),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: Colors.grey[200]!),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            'Net Hedefi Göstergesi',
            style: TextStyle(
              fontSize: 16,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 16),
          // Progress Bar
          Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Row(
                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text(
                    'Mevcut: ${currentNet.toStringAsFixed(2)}',
                    style: const TextStyle(fontSize: 12),
                  ),
                  Text(
                    'Hedef: ${targetNet.toStringAsFixed(2)}',
                    style: const TextStyle(fontSize: 12),
                  ),
                ],
              ),
              const SizedBox(height: 8),
              LinearProgressIndicator(
                value: progress,
                minHeight: 10,
                backgroundColor: Colors.grey[200],
                valueColor: AlwaysStoppedAnimation<Color>(
                  progress >= 1.0 ? Colors.green : const Color(0xFF4F46E5),
                ),
              ),
              const SizedBox(height: 8),
              Text(
                '${(progress * 100).toStringAsFixed(1)}% tamamlandı',
                style: TextStyle(
                  fontSize: 12,
                  color: Colors.grey[600],
                ),
              ),
            ],
          ),
          const SizedBox(height: 16),
          // Status Message
          if (progress >= 1.0)
            Container(
              padding: const EdgeInsets.all(12),
              decoration: BoxDecoration(
                color: Colors.green[50],
                borderRadius: BorderRadius.circular(8),
              ),
              child: Row(
                children: [
                  const Icon(Icons.check_circle, color: Colors.green),
                  const SizedBox(width: 8),
                  Expanded(
                    child: Text(
                      'Hedefine ulaştın! Başarılar diliyoruz! 🎉',
                      style: TextStyle(
                        color: Colors.green[700],
                        fontWeight: FontWeight.bold,
                      ),
                    ),
                  ),
                ],
              ),
            )
          else
            Container(
              padding: const EdgeInsets.all(12),
              decoration: BoxDecoration(
                color: const Color(0xFF4F46E5).withOpacity(0.1),
                borderRadius: BorderRadius.circular(8),
              ),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    'Hedefe Kalan: ${remaining.toStringAsFixed(2)} net',
                    style: const TextStyle(
                      fontWeight: FontWeight.bold,
                      color: Color(0xFF4F46E5),
                    ),
                  ),
                  const SizedBox(height: 4),
                  Text(
                    'Günde ${(remaining / (examDateInfo!.daysRemaining > 0 ? examDateInfo!.daysRemaining : 1)).toStringAsFixed(2)} net artış hedefle',
                    style: TextStyle(
                      fontSize: 12,
                      color: Colors.grey[600],
                    ),
                  ),
                ],
              ),
            ),
        ],
      ),
    );
  }
}

class _SubjectBarChart extends StatelessWidget {
  final List<SubjectAnalytics> subjectAnalytics;

  const _SubjectBarChart({required this.subjectAnalytics});

  @override
  Widget build(BuildContext context) {
    return Container(
      margin: const EdgeInsets.symmetric(horizontal: 16),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: Colors.white,
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: Colors.grey[200]!),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            'Ders Başarıları',
            style: TextStyle(
              fontSize: 16,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 16),
          SizedBox(
            height: 300,
            child: BarChart(
              BarChartData(
                alignment: BarChartAlignment.spaceAround,
                maxY: 100,
                barTouchData: BarTouchData(
                  enabled: true,
                  touchTooltipData: BarTouchTooltipData(
                    tooltipBgColor: Colors.grey[800],
                    getTooltipItem: (group, groupIndex, rod, rodIndex) {
                      return BarTooltipItem(
                        '${rod.toY.toStringAsFixed(1)}%',
                        const TextStyle(color: Colors.white),
                      );
                    },
                  ),
                ),
                gridData: FlGridData(
                  show: true,
                  drawVerticalLine: false,
                  horizontalInterval: 20,
                ),
                barGroups: subjectAnalytics.asMap().entries.map((entry) {
                  final index = entry.key;
                  final subject = entry.value;
                  return BarChartGroupData(
                    x: index,
                    barRods: [
                      BarChartRodData(
                        toY: subject.successPercentage,
                        color: _getSubjectColor(subject.successPercentage),
                        width: 12,
                      ),
                    ],
                  );
                }).toList(),
                titlesData: FlTitlesData(
                  show: true,
                  topTitles: AxisTitles(
                    sideTitles: SideTitles(showTitles: false),
                  ),
                  rightTitles: AxisTitles(
                    sideTitles: SideTitles(showTitles: false),
                  ),
                  bottomTitles: AxisTitles(
                    sideTitles: SideTitles(
                      showTitles: true,
                      getTitlesWidget: (value, meta) {
                        final index = value.toInt();
                        if (index < subjectAnalytics.length) {
                          return Padding(
                            padding: const EdgeInsets.only(top: 8),
                            child: Text(
                              subjectAnalytics[index]
                                  .subjectName
                                  .substring(0, 3),
                              style: const TextStyle(fontSize: 10),
                            ),
                          );
                        }
                        return const Text('');
                      },
                    ),
                  ),
                  leftTitles: AxisTitles(
                    sideTitles: SideTitles(
                      showTitles: true,
                      interval: 20,
                      getTitlesWidget: (value, meta) {
                        return Text(
                          '${value.toInt()}%',
                          style: const TextStyle(fontSize: 10),
                        );
                      },
                    ),
                  ),
                ),
              ),
            ),
          ),
        ],
      ),
    );
  }

  Color _getSubjectColor(double percentage) {
    if (percentage >= 80) return Colors.green;
    if (percentage >= 60) return Colors.blue;
    if (percentage >= 40) return Colors.orange;
    return Colors.red;
  }
}

                  drawBehindDataPoints: true,
                ),
                titlePositionPercentageOffset: 0.2,
                getTitle: (index, angle) {
                  if (index < subjectAnalytics.length) {
                    return RadarChartTitle(
                      text: subjectAnalytics[index].subjectName,
                      angle: angle,
                    );
                  }
                  return RadarChartTitle(text: '');
                },
                tickCount: 5,
                ticksTextStyle: const TextStyle(fontSize: 10),
                radarTouchData: RadarTouchData(enabled: true),
              ),
            ),
          ),
          const SizedBox(height: 16),
          // Legend
          Wrap(
            spacing: 16,
            runSpacing: 12,
            children: subjectAnalytics.map((subject) {
              return Row(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Container(
                    width: 12,
                    height: 12,
                    decoration: BoxDecoration(
                      color: _getPerformanceColor(subject.successPercentage),
                      shape: BoxShape.circle,
                    ),
                  ),
                  const SizedBox(width: 8),
                  Flexible(
                    child: Text(
                      '${subject.subjectName}: ${subject.successPercentage.toStringAsFixed(1)}%',
                      style: const TextStyle(fontSize: 12),
                    ),
                  ),
                ],
              );
            }).toList(),
          ),
        ],
      ),
    );
  }

  Color _getPerformanceColor(double percentage) {
    if (percentage >= 80) return Colors.green;
    if (percentage >= 60) return Colors.blue;
    if (percentage >= 40) return Colors.orange;
    return Colors.red;
  }
}

double? _parseDouble(dynamic value) {
  if (value == null) return null;
  if (value is double) return value;
  if (value is int) return value.toDouble();
  if (value is String) return double.tryParse(value);
  return null;
}
