/*
// android/app/src/main/kotlin/com/kpss/app/MainActivity.kt

package com.kpss.app

import android.app.NotificationChannel
import android.app.NotificationManager
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine

class MainActivity : FlutterActivity() {

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        createNotificationChannels()
    }

    private fun createNotificationChannels() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val notificationManager = getSystemService(NOTIFICATION_SERVICE) as NotificationManager

            // Channel 1: Günlük Bildirimler
            val dailyChannel = NotificationChannel(
                "kpss_daily",
                "Günlük Bildirimler",
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply {
                description = "Günlük çalışma hatırlatmaları"
                enableVibration(false)
                setSound(null, null)
            }

            // Channel 2: Genel Hatırlatmalar
            val remindersChannel = NotificationChannel(
                "kpss_reminders",
                "Hatırlatmalar",
                NotificationManager.IMPORTANCE_HIGH
            ).apply {
                description = "Test ve etkinlik hatırlatmaları"
                enableVibration(true)
            }

            // Channel 3: Acil Duyurular
            val urgentChannel = NotificationChannel(
                "kpss_urgent",
                "Acil Duyurular",
                NotificationManager.IMPORTANCE_MAX
            ).apply {
                description = "Acil bildirimler"
                enableVibration(true)
            }

            notificationManager.apply {
                createNotificationChannel(dailyChannel)
                createNotificationChannel(remindersChannel)
                createNotificationChannel(urgentChannel)
            }
        }
    }
}
*/

/*
1. ENV SETUP:
   - .env.example'ı .env olarak kopyala
   - Supabase ve Firebase credentials'ını ekle
   - export $(cat .env | grep -v '#' | xargs)

2. SUPABASE SETUP:
   - PostgreSQL schema'yı Supabase console'da çalıştır (01_supabase_schema.sql)
   - RLS politikalarını doğrula
   - Service role key'i backend için kaydet

3. FIREBASE SETUP:
   - Firebase console'da yeni proje oluştur
   - AndroidManifest.xml'e sender ID ekle
   - GoogleService-Info.plist'i ios/Runner/'a ekle
   - google-services.json'u android/app/'a ekle

4. DEPENDENCIES:
   - flutter pub get
   - cd ios && pod install && cd ..

5. BUILD & RUN:
   - Android: flutter run -d android
   - iOS: flutter run -d ios
   - Web: flutter run -d chrome

6. TESTING:
   - flutter test
   - flutter test --coverage

7. BUILD:
   - Android APK: flutter build apk
   - Android App Bundle: flutter build appbundle
   - iOS IPA: flutter build ios

8. DEPLOYMENT:
   - Play Store: Upload appbundle ke Play Console
   - App Store: Upload dengan App Store Connect
   - TestFlight: Beta testing için upload
*/
