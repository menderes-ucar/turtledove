class ErrorNotebookScreen extends StatefulWidget {
  const ErrorNotebookScreen({Key? key}) : super(key: key);

  @override
  State<ErrorNotebookScreen> createState() => _ErrorNotebookScreenState();
}

class _ErrorNotebookScreenState extends State<ErrorNotebookScreen>
    with SingleTickerProviderStateMixin {
  late TabController _tabController;
  List<String> _subjectIds = [];
  Map<String, List<ErrorNotebookItemModel>> _errorsBySubject = {};
  bool _isLoading = false;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: 1, vsync: this);
    WidgetsBinding.instance.addPostFrameCallback((_) {
      _loadErrorNotebook();
    });
  }

  Future<void> _loadErrorNotebook() async {
    setState(() {
      _isLoading = true;
      _errorMessage = null;
    });

    try {
      // TODO: Supabase'den verileri yükle
      // Bu örnek için mock data kullanıyoruz
      _subjectIds = ['subject_1', 'subject_2', 'subject_3'];
      _errorsBySubject = {
        'Tarih': [
          ErrorNotebookItemModel(
            id: '1',
            userId: 'user_1',
            questionId: 'q1',
            resolved: false,
            wrongCount: 3,
            lastWrongAt: DateTime.now().subtract(const Duration(days: 2)),
            noteText: 'Osmanlı dönemini karıştırıyorum',
            createdAt: DateTime.now().subtract(const Duration(days: 5)),
            updatedAt: DateTime.now().subtract(const Duration(days: 2)),
          ),
        ],
        'Coğrafya': [
          ErrorNotebookItemModel(
            id: '2',
            userId: 'user_1',
            questionId: 'q2',
            resolved: false,
            wrongCount: 2,
            lastWrongAt: DateTime.now().subtract(const Duration(days: 1)),
            noteText: null,
            createdAt: DateTime.now().subtract(const Duration(days: 3)),
            updatedAt: DateTime.now().subtract(const Duration(days: 1)),
          ),
        ],
      };

      if (_subjectIds.isNotEmpty) {
        _tabController = TabController(
          length: _subjectIds.length,
          vsync: this,
        );
      }

      _isLoading = false;
      if (mounted) setState(() {});
    } catch (e) {
      _errorMessage = 'Yanlış defteri yüklenemedi: ${e.toString()}';
      _isLoading = false;
      if (mounted) setState(() {});
    }
  }

  @override
  void dispose() {
    _tabController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.grey[50],
      appBar: AppBar(
        title: const Text('Yanlış Defteri'),
        centerTitle: true,
        elevation: 0,
        backgroundColor: Colors.white,
        foregroundColor: Colors.black,
      ),
      body: _isLoading
          ? Center(
              child: LoadingAnimationWidget.discreteCircle(
                color: const Color(0xFF4F46E5),
                size: 50,
              ),
            )
          : _errorMessage != null
              ? Center(
                  child: Column(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      const Icon(
                        Icons.error_outline,
                        size: 64,
                        color: Colors.red,
                      ),
                      const SizedBox(height: 16),
                      Text(_errorMessage!),
                      const SizedBox(height: 24),
                      ElevatedButton(
                        onPressed: _loadErrorNotebook,
                        child: const Text('Yeniden Dene'),
                      ),
                    ],
                  ),
                )
              : Column(
                  children: [
                    // Sekmeler
                    if (_subjectIds.isNotEmpty)
                      Container(
                        color: Colors.white,
                        child: TabBar(
                          controller: _tabController,
                          isScrollable: true,
                          labelColor: const Color(0xFF4F46E5),
                          unselectedLabelColor: Colors.grey[600],
                          indicatorColor: const Color(0xFF4F46E5),
                          indicatorWeight: 3,
                          tabs: _subjectIds
                              .map((subjectId) => Tab(text: subjectId))
                              .toList(),
                        ),
                      ),
                    // Tab İçeriği
                    Expanded(
                      child: TabBarView(
                        controller: _tabController,
                        children: _subjectIds
                            .map((subjectId) => _ErrorListView(
                                  errors: _errorsBySubject[subjectId] ?? [],
                                  subjectId: subjectId,
                                  onRetakeQuiz: () => _startRemediationQuiz(subjectId),
                                ))
                            .toList(),
                      ),
                    ),
                  ],
                ),
    );
  }

  void _startRemediationQuiz(String subjectId) {
    // Quiz Provider'ı error notebook soruları ile başlat
    final quizProvider = context.read<QuizProvider>();
    
    Navigator.of(context).push(
      MaterialPageRoute(
        builder: (context) => QuizScreen(
          subjectId: subjectId,
          fromErrorNotebook: true,
        ),
      ),
    );
  }
}

// ============================================================================
// ERROR LIST VIEW
// ============================================================================

class _ErrorListView extends StatelessWidget {
  final List<ErrorNotebookItemModel> errors;
  final String subjectId;
  final VoidCallback onRetakeQuiz;

  const _ErrorListView({
    required this.errors,
    required this.subjectId,
    required this.onRetakeQuiz,
  });

  @override
  Widget build(BuildContext context) {
    final unresolvedErrors = errors.where((e) => !e.resolved).toList();

    if (unresolvedErrors.isEmpty) {
      return Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(
              Icons.check_circle,
              size: 80,
              color: Colors.green[300],
            ),
            const SizedBox(height: 16),
            Text(
              'Bu konuda yanlış yapılmış soru yok',
              style: TextStyle(
                fontSize: 18,
                color: Colors.grey[600],
              ),
            ),
          ],
        ),
      );
    }

    return ListView.builder(
      padding: const EdgeInsets.all(16),
      itemCount: unresolvedErrors.length + 1,
      itemBuilder: (context, index) {
        if (index == unresolvedErrors.length) {
          return Padding(
            padding: const EdgeInsets.only(top: 16),
            child: _RemediationTestButton(
              errorCount: unresolvedErrors.length,
              onTap: onRetakeQuiz,
            ),
          );
        }

        final error = unresolvedErrors[index];
        return Padding(
          padding: const EdgeInsets.only(bottom: 12),
          child: _ErrorItemCard(
            error: error,
            onMarkResolved: () {
              // Mark as resolved
              _markErrorAsResolved(error.id);
            },
          ),
        );
      },
    );
  }

  void _markErrorAsResolved(String errorId) {
    // TODO: Supabase'de mark as resolved
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Soru çözülmüş olarak işaretlendi')),
    );
  }
}

class _ErrorItemCard extends StatefulWidget {
  final ErrorNotebookItemModel error;
  final VoidCallback onMarkResolved;

  const _ErrorItemCard({
    required this.error,
    required this.onMarkResolved,
  });

  @override
  State<_ErrorItemCard> createState() => _ErrorItemCardState();
}

class _ErrorItemCardState extends State<_ErrorItemCard> {
  bool _isExpanded = false;

  @override
  Widget build(BuildContext context) {
    return Card(
      elevation: 0,
      color: Colors.white,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(12),
        side: BorderSide(
          color: _getErrorLevelColor(widget.error.wrongCount).withOpacity(0.3),
          width: 2,
        ),
      ),
      child: InkWell(
        onTap: () {
          setState(() {
            _isExpanded = !_isExpanded;
          });
        },
        borderRadius: BorderRadius.circular(12),
        child: Column(
          children: [
            Padding(
              padding: const EdgeInsets.all(16),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  // Başlık ve Hata Seviyesi
                  Row(
                    children: [
                      Expanded(
                        child: Column(
                          crossAxisAlignment: CrossAxisAlignment.start,
                          children: [
                            Text(
                              'Soru ${widget.error.questionId}',
                              style: const TextStyle(
                                fontSize: 16,
                                fontWeight: FontWeight.bold,
                              ),
                            ),
                            const SizedBox(height: 4),
                            Text(
                              'Hata Sayısı: ${widget.error.wrongCount}',
                              style: TextStyle(
                                fontSize: 13,
                                color: Colors.grey[600],
                              ),
                            ),
                          ],
                        ),
                      ),
                      // Hata Seviyesi Badge
                      Container(
                        padding: const EdgeInsets.symmetric(
                          horizontal: 12,
                          vertical: 6,
                        ),
                        decoration: BoxDecoration(
                          color: _getErrorLevelColor(widget.error.wrongCount)
                              .withOpacity(0.2),
                          borderRadius: BorderRadius.circular(20),
                        ),
                        child: Text(
                          _getErrorLevelLabel(widget.error.wrongCount),
                          style: TextStyle(
                            fontSize: 12,
                            fontWeight: FontWeight.bold,
                            color: _getErrorLevelColor(widget.error.wrongCount),
                          ),
                        ),
                      ),
                    ],
                  ),
                  const SizedBox(height: 12),
                  // Son Hata Tarihi
                  Row(
                    children: [
                      Icon(
                        Icons.access_time,
                        size: 16,
                        color: Colors.grey[600],
                      ),
                      const SizedBox(width: 8),
                      Text(
                        'Son hata: ${_formatDate(widget.error.lastWrongAt)}',
                        style: TextStyle(
                          fontSize: 12,
                          color: Colors.grey[600],
                        ),
                      ),
                    ],
                  ),
                ],
              ),
            ),
            // Genişletilmiş İçerik
            if (_isExpanded) ...[
              const Divider(height: 0),
              Padding(
                padding: const EdgeInsets.all(16),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    if (widget.error.noteText != null &&
                        widget.error.noteText!.isNotEmpty) ...[
                      const Text(
                        'Notlar',
                        style: TextStyle(
                          fontSize: 14,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                      const SizedBox(height: 8),
                      Container(
                        padding: const EdgeInsets.all(12),
                        decoration: BoxDecoration(
                          color: Colors.grey[100],
                          borderRadius: BorderRadius.circular(8),
                        ),
                        child: Text(
                          widget.error.noteText!,
                          style: TextStyle(
                            fontSize: 13,
                            color: Colors.grey[800],
                            height: 1.5,
                          ),
                        ),
                      ),
                      const SizedBox(height: 12),
                    ],
                    // Butonlar
                    Row(
                      mainAxisAlignment: MainAxisAlignment.spaceBetween,
                      children: [
                        OutlinedButton.icon(
                          onPressed: () => _editNote(),
                          icon: const Icon(Icons.edit),
                          label: const Text('Not Ekle/Düzenle'),
                        ),
                        ElevatedButton.icon(
                          onPressed: widget.onMarkResolved,
                          icon: const Icon(Icons.check),
                          label: const Text('Çözüldü'),
                          style: ElevatedButton.styleFrom(
                            backgroundColor: Colors.green,
                          ),
                        ),
                      ],
                    ),
                  ],
                ),
              ),
            ],
          ],
        ),
      ),
    );
  }

  void _editNote() {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      builder: (context) => _AddNoteBottomSheet(
        initialNote: widget.error.noteText,
        onSave: (note) {
          // TODO: Supabase'de notu güncelle
          Navigator.of(context).pop();
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('Not kaydedildi')),
          );
        },
      ),
    );
  }

  Color _getErrorLevelColor(int wrongCount) {
    if (wrongCount >= 5) return Colors.red;
    if (wrongCount >= 3) return Colors.orange;
    return Colors.yellow[700] ?? Colors.yellow;
  }

  String _getErrorLevelLabel(int wrongCount) {
    if (wrongCount >= 5) return 'Kritik';
    if (wrongCount >= 3) return 'Orta';
    return 'Hafif';
  }

  String _formatDate(DateTime date) {
    final now = DateTime.now();
    final difference = now.difference(date);

    if (difference.inDays == 0) {
      return 'Bugün';
    } else if (difference.inDays == 1) {
      return 'Dün';
    } else if (difference.inDays < 7) {
      return '${difference.inDays} gün önce';
    } else {
      return intl.DateFormat('dd.MM.yyyy').format(date);
    }
  }
}

class _AddNoteBottomSheet extends StatefulWidget {
  final String? initialNote;
  final Function(String) onSave;

  const _AddNoteBottomSheet({
    required this.initialNote,
    required this.onSave,
  });

  @override
  State<_AddNoteBottomSheet> createState() => _AddNoteBottomSheetState();
}

class _AddNoteBottomSheetState extends State<_AddNoteBottomSheet> {
  late TextEditingController _noteController;

  @override
  void initState() {
    super.initState();
    _noteController = TextEditingController(
      text: widget.initialNote ?? '',
    );
  }

  @override
  void dispose() {
    _noteController.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      padding: EdgeInsets.only(
        left: 16,
        right: 16,
        top: 16,
        bottom: MediaQuery.of(context).viewInsets.bottom + 16,
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            'Soru Notu',
            style: TextStyle(
              fontSize: 18,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 16),
          TextField(
            controller: _noteController,
            maxLines: 4,
            decoration: InputDecoration(
              hintText: 'Bu soruda neyi öğrenmek istediğini yaz...',
              border: OutlineInputBorder(
                borderRadius: BorderRadius.circular(8),
              ),
            ),
          ),
          const SizedBox(height: 16),
          SizedBox(
            width: double.infinity,
            child: ElevatedButton(
              onPressed: () {
                widget.onSave(_noteController.text);
              },
              child: const Text('Kaydet'),
            ),
          ),
        ],
      ),
    );
  }
}
