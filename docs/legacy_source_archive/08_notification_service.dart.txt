import 'dart:io';
import 'package:firebase_core/firebase_core.dart';
import 'package:firebase_messaging/firebase_messaging.dart';
import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

// Import models
// import '../models/dart_models.dart';

// ============================================================================
// NOTIFICATION SERVICE (SINGLETON)
// ============================================================================

class NotificationService {
  static final NotificationService _instance = NotificationService._internal();

  factory NotificationService() {
    return _instance;
  }

  NotificationService._internal();

  // FCM Instance
  late FirebaseMessaging _firebaseMessaging;
  
  // Local Notifications
  late FlutterLocalNotificationsPlugin _localNotifications;
  
  // Navigation Callback
  Function(String? route)? onNotificationTapped;
  
  // State
  String? _fcmToken;
  bool _isInitialized = false;

  // Getters
  String? get fcmToken => _fcmToken;
  bool get isInitialized => _isInitialized;

  // =========================================================================
  // INITIALIZATION
  // =========================================================================

  Future<void> initialize({
    required SupabaseClient supabaseClient,
    required String userId,
    required Function(String? route) onNotificationHandler,
  }) async {
    if (_isInitialized) return;

    try {
      _firebaseMessaging = FirebaseMessaging.instance;
      onNotificationTapped = onNotificationHandler;

      // Platform-spesifik ayarlar
      if (Platform.isIOS) {
        await _setupiOS();
      } else if (Platform.isAndroid) {
        await _setupAndroid();
      }

      // Local Notifications Setup
      await _setupLocalNotifications();

      // FCM Token Al
      await _retrieveAndSaveToken(supabaseClient, userId);

      // Foreground Bildirim Listener
      FirebaseMessaging.onMessage.listen((RemoteMessage message) {
        _handleForegroundNotification(message);
      });

      // Background/Terminated Notification Listener
      FirebaseMessaging.onMessageOpenedApp.listen((RemoteMessage message) {
        _handleNotificationTap(message);
      });

      // Token Refresh Listener
      _firebaseMessaging.onTokenRefresh.listen((newToken) async {
        _fcmToken = newToken;
        await _saveTokenToSupabase(supabaseClient, userId, newToken);
      });

      _isInitialized = true;
      print('✅ NotificationService başarıyla initialized');
    } catch (e) {
      print('❌ NotificationService initialization hatası: $e');
      rethrow;
    }
  }

  // =========================================================================
  // iOS SETUP
  // =========================================================================

  Future<void> _setupiOS() async {
    try {
      final settings = await _firebaseMessaging.requestPermission(
        alert: true,
        announcement: true,
        badge: true,
        carPlay: false,
        criticalAlert: false,
        provisional: false,
        sound: true,
      );

      print('iOS Notification Permission Status: ${settings.authorizationStatus}');

      // APNs token al (iOS için gerekli)
      final apnsToken = await _firebaseMessaging.getAPNSToken();
      print('APNS Token: $apnsToken');

      // APNs token yoksa warning
      if (apnsToken == null) {
        print('⚠️ APNS Token alınamadı. Bildirimler gösterilmeyebilir.');
      }
    } catch (e) {
      print('iOS Setup Hatası: $e');
    }
  }

  // =========================================================================
  // ANDROID SETUP
  // =========================================================================

  Future<void> _setupAndroid() async {
    try {
      if (Platform.isAndroid) {
        // Notification Channel Oluştur: Günlük Bildirimler
        await _createNotificationChannel(
          id: 'kpss_daily',
          name: 'Günlük Bildirimler',
          description: 'Günlük çalışma hatırlatmaları',
          importance: Importance.defaultImportance,
        );

        // Notification Channel Oluştur: Genel Hatırlatmalar
        await _createNotificationChannel(
          id: 'kpss_reminders',
          name: 'Hatırlatmalar',
          description: 'Test ve etkinlik hatırlatmaları',
          importance: Importance.highImportance,
        );

        // Notification Channel Oluştur: Acil Duyurular
        await _createNotificationChannel(
          id: 'kpss_urgent',
          name: 'Acil Duyurular',
          description: 'Acil bildirimler',
          importance: Importance.maxImportance,
          enableVibration: true,
        );
      }

      // Permission İste
      final notificationSettings = await _firebaseMessaging.requestPermission(
        alert: true,
        badge: true,
        provisional: true,
        sound: true,
      );

      print('Android Permission Status: ${notificationSettings.authorizationStatus}');
    } catch (e) {
      print('Android Setup Hatası: $e');
    }
  }

  // =========================================================================
  // LOCAL NOTIFICATIONS SETUP
  // =========================================================================

  Future<void> _setupLocalNotifications() async {
    try {
      _localNotifications = FlutterLocalNotificationsPlugin();

      // Android Settings
      const AndroidInitializationSettings androidSettings =
          AndroidInitializationSettings('@mipmap/launcher_icon');

      // iOS Settings
      const DarwinInitializationSettings iosSettings =
          DarwinInitializationSettings(
        requestAlertPermission: true,
        requestBadgePermission: true,
        requestSoundPermission: true,
      );

      // Genel Settings
      final InitializationSettings initSettings = InitializationSettings(
        android: androidSettings,
        iOS: iosSettings,
      );

      // Initialize
      await _localNotifications.initialize(
        initSettings,
        onDidReceiveNotificationResponse: (NotificationResponse response) {
          _handleLocalNotificationTap(response.payload);
        },
      );

      print('✅ Local Notifications setup tamamlandı');
    } catch (e) {
      print('Local Notifications Setup Hatası: $e');
    }
  }

  // =========================================================================
  // NOTIFICATION CHANNEL OLUŞTURMA (ANDROID)
  // =========================================================================

  Future<void> _createNotificationChannel({
    required String id,
    required String name,
    required String description,
    required Importance importance,
    bool enableVibration = false,
  }) async {
    try {
      final androidChannel = AndroidNotificationChannel(
        id,
        name,
        description: description,
        importance: importance,
        enableVibration: enableVibration,
        playSound: true,
        soundSource: AndroidNotificationSoundSource.defaultRingtone,
      );

      await _localNotifications
          .resolvePlatformSpecificImplementation<
              AndroidFlutterLocalNotificationsPlugin>()
          ?.createNotificationChannel(androidChannel);

      print('✅ Notification Channel oluşturuldu: $id');
    } catch (e) {
      print('Channel Oluşturma Hatası: $e');
    }
  }

  // =========================================================================
  // FCM TOKEN YÖNETİMİ
  // =========================================================================

  Future<void> _retrieveAndSaveToken(
    SupabaseClient supabaseClient,
    String userId,
  ) async {
    try {
      final token = await _firebaseMessaging.getToken();
      
      if (token != null) {
        _fcmToken = token;
        await _saveTokenToSupabase(supabaseClient, userId, token);
        print('✅ FCM Token alındı ve kaydedildi');
      } else {
        print('⚠️ FCM Token alınamadı');
      }
    } catch (e) {
      print('FCM Token Retrieval Hatası: $e');
    }
  }

  Future<void> _saveTokenToSupabase(
    SupabaseClient supabaseClient,
    String userId,
    String token,
  ) async {
    try {
      final platform = Platform.isIOS ? 'ios' : 'android';
      final deviceName = _getDeviceName();

      await supabaseClient.from('user_devices').upsert({
        'user_id': userId,
        'fcm_token': token,
        'platform': platform,
        'device_name': deviceName,
        'updated_at': DateTime.now().toIso8601String(),
      }, onConflict: 'user_id,fcm_token');

      print('✅ Token Supabase\'e kaydedildi');
    } catch (e) {
      print('Token Kayıt Hatası: $e');
    }
  }

  String _getDeviceName() {
    if (Platform.isIOS) {
      return 'iOS Device';
    } else if (Platform.isAndroid) {
      return 'Android Device';
    }
    return 'Unknown Device';
  }

  // =========================================================================
  // FOREGROUND NOTIFICATION HANDLING
  // =========================================================================

  void _handleForegroundNotification(RemoteMessage message) {
    try {
      final notification = message.notification;
      final data = message.data;

      print('📨 Foreground Bildirim: ${notification?.title}');

      // Local notification göster
      _showLocalNotification(
        title: notification?.title ?? 'KPSS',
        body: notification?.body ?? '',
        payload: _extractRoute(data),
      );
    } catch (e) {
      print('Foreground Notification Handling Hatası: $e');
    }
  }

  Future<void> _showLocalNotification({
    required String title,
    required String body,
    String? payload,
  }) async {
    try {
      const AndroidNotificationDetails androidDetails =
          AndroidNotificationDetails(
        'kpss_daily',
        'Günlük Bildirimler',
        importance: Importance.defaultImportance,
        priority: Priority.defaultPriority,
        showWhen: true,
      );

      const DarwinNotificationDetails iosDetails =
          DarwinNotificationDetails();

      const NotificationDetails details = NotificationDetails(
        android: androidDetails,
        iOS: iosDetails,
      );

      await _localNotifications.show(
        DateTime.now().millisecond,
        title,
        body,
        details,
        payload: payload,
      );

      print('✅ Local notification gösterildi');
    } catch (e) {
      print('Local Notification Show Hatası: $e');
    }
  }

  // =========================================================================
  // NOTIFICATION TAP HANDLING
  // =========================================================================

  void _handleNotificationTap(RemoteMessage message) {
    try {
      final route = _extractRoute(message.data);
      print('🔗 Bildirime tıklandı. Route: $route');
      onNotificationTapped?.call(route);
    } catch (e) {
      print('Notification Tap Handling Hatası: $e');
    }
  }

  void _handleLocalNotificationTap(String? payload) {
    try {
      print('🔗 Local bildirime tıklandı. Payload: $payload');
      onNotificationTapped?.call(payload);
    } catch (e) {
      print('Local Notification Tap Handling Hatası: $e');
    }
  }

  // =========================================================================
  // ROUTE EXTRACTION
  // =========================================================================

  String? _extractRoute(Map<String, dynamic> data) {
    final type = data['type'] as String?;
    
    switch (type) {
      case 'quiz':
        return '/quiz/${data['subject_id']}';
      case 'error_notebook':
        return '/error_notebook';
      case 'study_plan':
        return '/study_plan';
      case 'notes':
        return '/notes';
      default:
        return null;
    }
  }

  // =========================================================================
  // NOTIFICATION GÖNDERME (Backend'den tetiklenir)
  // =========================================================================

  static Future<void> sendNotification({
    required String title,
    required String body,
    required String? fcmToken,
    Map<String, String>? data,
  }) async {
    if (fcmToken == null) {
      print('⚠️ FCM Token boş, bildirim gönderilemedi');
      return;
    }

    // Bu fonksiyon genellikle backend tarafından çalıştırılır
    // Flutter tarafında test için kullanılabilir
    print('📤 Bildirim gönderiliyor: $title');
  }

  // =========================================================================
  // CLEANUP
  // =========================================================================

  Future<void> dispose() async {
    try {
      _isInitialized = false;
      print('🧹 NotificationService disposed');
    } catch (e) {
      print('Dispose Hatası: $e');
    }
  }
}

// ============================================================================
// FCM BACKGROUND HANDLER (App'in dışında çalışan)
// main.dart'da tetiklenmesi gerekir
// ============================================================================

Future<void> firebaseMessagingBackgroundHandler(RemoteMessage message) async {
  // Uygulamanın kapalı olduğu durumlarda çalışır
  try {
    print('🔔 Background Bildirim: ${message.notification?.title}');
    
    // Gerekirse burada veritabanına işlem yap
    // Ama Flutter widgets'lara erişim yok
  } catch (e) {
    print('Background Handler Hatası: $e');
  }
}
