Future<void> _firebaseMessagingBackgroundHandler(RemoteMessage message) async {
  await Firebase.initializeApp();
  print('Handling a background message: ${message.messageId}');
}

// ============================================================================
// MAIN FUNCTION
// ============================================================================

Future<void> main() async {
  // Flutter Binding
  WidgetsFlutterBinding.ensureInitialized();

  try {
    // Supabase İnitialize
    await Supabase.initialize(
      url: const String.fromEnvironment('SUPABASE_URL'),
      anonKey: const String.fromEnvironment('SUPABASE_ANON_KEY'),
    );

    // Firebase İnitialize
    await Firebase.initializeApp(
      options: const FirebaseOptions(
        apiKey: String.fromEnvironment('FIREBASE_API_KEY'),
        appId: String.fromEnvironment('FIREBASE_APP_ID'),
        messagingSenderId: String.fromEnvironment('FIREBASE_MESSAGING_SENDER_ID'),
        projectId: String.fromEnvironment('FIREBASE_PROJECT_ID'),
        storageBucket: String.fromEnvironment('FIREBASE_STORAGE_BUCKET'),
      ),
    );

    // Background Message Handler
    FirebaseMessaging.onBackgroundMessage(_firebaseMessagingBackgroundHandler);

    print('✅ Supabase ve Firebase başarıyla initialize edildi');
  } catch (e) {
    print('❌ İnitialization Hatası: $e');
    rethrow;
  }

  runApp(const KpssApp());
}

class KpssApp extends StatefulWidget {
  const KpssApp({Key? key}) : super(key: key);

  @override
  State<KpssApp> createState() => _KpssAppState();
}

class _KpssAppState extends State<KpssApp> {
  late final GoRouter _router;
  final GlobalKey<NavigatorState> _rootNavigatorKey =
      GlobalKey<NavigatorState>();
  late NotificationService _notificationService;

  @override
  void initState() {
    super.initState();
    _initializeNotifications();
    _setupRouter();
  }

  void _initializeNotifications() {
    _notificationService = NotificationService();
    
    // Get current user
    final supabaseClient = Supabase.instance.client;
    final userId = supabaseClient.auth.currentUser?.id;

    if (userId != null) {
      _notificationService.initialize(
        supabaseClient: supabaseClient,
        userId: userId,
        onNotificationHandler: _handleNotificationNavigation,
      );
    }
  }

  void _handleNotificationNavigation(String? route) {
    if (route != null) {
      _rootNavigatorKey.currentState?.pushNamed(route);
    }
  }

  void _setupRouter() {
    _router = GoRouter(
      navigatorKey: _rootNavigatorKey,
      initialLocation: '/',
      routes: [
        // Ana Sayfa
        GoRoute(
          path: '/',
          name: 'home',
          builder: (context, state) => const HomeScreen(),
          routes: [
            // Quiz Ekranı
            GoRoute(
              path: 'quiz/:subjectId',
              name: 'quiz',
              builder: (context, state) {
                final subjectId = state.pathParameters['subjectId']!;
                final fromErrorNotebook =
                    state.queryParameters['fromErrorNotebook'] == 'true';
                return QuizScreen(
                  subjectId: subjectId,
                  fromErrorNotebook: fromErrorNotebook,
                );
              },
            ),
            // Yanlış Defteri
            GoRoute(
              path: 'error-notebook',
              name: 'error_notebook',
              builder: (context, state) => const ErrorNotebookScreen(),
            ),
            // Notlar
            GoRoute(
              path: 'notes',
              name: 'notes',
              builder: (context, state) {
                final subjectId = state.queryParameters['subjectId'];
                return NotesScreen(initialSubjectId: subjectId);
              },
            ),
            // Çalışma Planı (Opsiyonel)
            GoRoute(
              path: 'study-plan',
              name: 'study_plan',
              builder: (context, state) => const StudyPlanScreen(),
            ),
            // Ayarlar (Opsiyonel)
            GoRoute(
              path: 'settings',
              name: 'settings',
              builder: (context, state) => const SettingsScreen(),
            ),
          ],
        ),
        // Login (Redirect'lere ihtiyaç varsa)
        GoRoute(
          path: '/login',
          name: 'login',
          builder: (context, state) => const LoginScreen(),
        ),
        // Signup (Opsiyonel)
        GoRoute(
          path: '/signup',
          name: 'signup',
          builder: (context, state) => const SignupScreen(),
        ),
      ],
      errorBuilder: (context, state) => ErrorScreen(error: state.error),
      redirect: (context, state) {
        // Auth redirect logic
        final isLoggedIn = Supabase.instance.client.auth.currentUser != null;
        final isLoggingIn = state.matchedLocation == '/login' ||
            state.matchedLocation == '/signup';

        if (!isLoggedIn && !isLoggingIn) {
          return '/login';
        }

        if (isLoggedIn && isLoggingIn) {
          return '/';
        }

        return null;
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    return MultiProvider(
      providers: _buildProviders(),
      child: MaterialApp.router(
        title: 'KPSS Kolay',
        routerConfig: _router,
        theme: _buildLightTheme(),
        darkTheme: _buildDarkTheme(),
        themeMode: ThemeMode.light,
        debugShowCheckedModeBanner: false,
        localizationsDelegates: const [],
        supportedLocales: const [Locale('tr', 'TR')],
      ),
    );
  }

  // =========================================================================
  // PROVIDERS SETUP
  // =========================================================================

  List<ChangeNotifierProvider> _buildProviders() {
    final supabaseClient = Supabase.instance.client;
    final userId = supabaseClient.auth.currentUser?.id ?? '';

    return [
      // Quiz Provider
      ChangeNotifierProvider(
        create: (_) => QuizProvider(
          supabaseClient: supabaseClient,
          userId: userId,
        ),
      ),
      // Notes and Audio Provider
      ChangeNotifierProvider(
        create: (_) => NotesAndAudioProvider(
          supabaseClient: supabaseClient,
          userId: userId,
        ),
      ),
      // Study Plan Provider
      ChangeNotifierProvider(
        create: (_) => StudyPlanProvider(
          supabaseClient: supabaseClient,
          userId: userId,
        ),
      ),
    ];
  }
