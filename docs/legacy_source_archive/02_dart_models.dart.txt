enum ExamType {
  lisans,
  onlisans,
  ortaogretim;

  String get displayName {
    switch (this) {
      case ExamType.lisans:
        return 'Lisans';
      case ExamType.onlisans:
        return 'Ön Lisans';
      case ExamType.ortaogretim:
        return 'Ortaöğretim';
    }
  }

  static ExamType fromString(String value) {
    return ExamType.values.firstWhere(
      (e) => e.name == value,
      orElse: () => ExamType.lisans,
    );
  }
}

enum QuizStatus {
  running,
  completed,
  abandoned;

  static QuizStatus fromString(String value) {
    return QuizStatus.values.firstWhere(
      (e) => e.name == value,
      orElse: () => QuizStatus.running,
    );
  }

  bool get isRunning => this == QuizStatus.running;
  bool get isCompleted => this == QuizStatus.completed;
  bool get isAbandoned => this == QuizStatus.abandoned;
}

@immutable
class QuestionModel {
  final String id;
  final String subjectId;
  final String questionText;
  final String? imageUrl;
  final String optionA;
  final String optionB;
  final String optionC;
  final String optionD;
  final String? optionE;
  final String? correctOption; // Server response only; never required in public question payload
  final String? explanation;
  final int difficultyLevel; // 1-5
  final String? source;
  final int? year;

  const QuestionModel({
    required this.id,
    required this.subjectId,
    required this.questionText,
    this.imageUrl,
    required this.optionA,
    required this.optionB,
    required this.optionC,
    required this.optionD,
    this.optionE,
    required this.correctOption,
    this.explanation,
    required this.difficultyLevel,
    this.source,
    this.year,
  });

  // Seçeneği metin olarak al
  String getOptionText(String option) {
    switch (option.toUpperCase()) {
      case 'A':
        return optionA;
      case 'B':
        return optionB;
      case 'C':
        return optionC;
      case 'D':
        return optionD;
      case 'E':
        return optionE ?? '';
      default:
        return '';
    }
  }

  // Zorluk seviyesi etiketini al
  String getDifficultyLabel() {
    switch (difficultyLevel) {
      case 1:
        return 'Çok Kolay';
      case 2:
        return 'Kolay';
      case 3:
        return 'Orta';
      case 4:
        return 'Zor';
      case 5:
        return 'Çok Zor';
      default:
        return 'Bilinmiyor';
    }
  }

  factory QuestionModel.fromJson(Map<String, dynamic> json) {
    return QuestionModel(
      id: json['id'] as String? ?? '',
      subjectId: json['subject_id'] as String? ?? '',
      questionText: json['question_text'] as String? ?? '',
      imageUrl: json['image_url'] as String?,
      optionA: json['option_a'] as String? ?? '',
      optionB: json['option_b'] as String? ?? '',
      optionC: json['option_c'] as String? ?? '',
      optionD: json['option_d'] as String? ?? '',
      optionE: json['option_e'] as String?,
      correctOption: (json['correct_option'] as String?)?.toUpperCase(),
      explanation: json['explanation'] as String?,
      difficultyLevel: json['difficulty_level'] as int? ?? 3,
      source: json['source'] as String?,
      year: json['year'] as int?,
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'subject_id': subjectId,
      'question_text': questionText,
      'image_url': imageUrl,
      'option_a': optionA,
      'option_b': optionB,
      'option_c': optionC,
      'option_d': optionD,
      'option_e': optionE,
      if (correctOption != null) 'correct_option': correctOption,
      'explanation': explanation,
      'difficulty_level': difficultyLevel,
      'source': source,
      'year': year,
    };
  }

  QuestionModel copyWith({
    String? id,
    String? subjectId,
    String? questionText,
    String? imageUrl,
    String? optionA,
    String? optionB,
    String? optionC,
    String? optionD,
    String? optionE,
    String? correctOption,
    String? explanation,
    int? difficultyLevel,
    String? source,
    int? year,
  }) {
    return QuestionModel(
      id: id ?? this.id,
      subjectId: subjectId ?? this.subjectId,
      questionText: questionText ?? this.questionText,
      imageUrl: imageUrl ?? this.imageUrl,
      optionA: optionA ?? this.optionA,
      optionB: optionB ?? this.optionB,
      optionC: optionC ?? this.optionC,
      optionD: optionD ?? this.optionD,
      optionE: optionE ?? this.optionE,
      correctOption: correctOption ?? this.correctOption,
      explanation: explanation ?? this.explanation,
      difficultyLevel: difficultyLevel ?? this.difficultyLevel,
      source: source ?? this.source,
      year: year ?? this.year,
    );
  }

  @override
  bool operator ==(Object other) =>
      identical(this, other) ||
      other is QuestionModel &&
          runtimeType == other.runtimeType &&
          id == other.id;

  @override
  int get hashCode => id.hashCode;

  @override
  String toString() => 'QuestionModel(id: $id)';
}

@immutable
class QuizAttemptModel {
  final String id;
  final String userId;
  final String subjectId;
  final QuizStatus status;
  final int totalQuestions;
  final int correctAnswers;
  final int wrongAnswers;
  final int blankAnswers;
  final int durationSeconds;
  final double? score;
  final bool isTimed;
  final int? timeLimitSeconds;
  final DateTime startedAt;
  final DateTime? completedAt;

  const QuizAttemptModel({
    required this.id,
    required this.userId,
    required this.subjectId,
    required this.status,
    required this.totalQuestions,
    required this.correctAnswers,
    required this.wrongAnswers,
    required this.blankAnswers,
    required this.durationSeconds,
    this.score,
    required this.isTimed,
    this.timeLimitSeconds,
    required this.startedAt,
    this.completedAt,
  });

  // Başarı yüzdesini hesapla
  double get successPercentage {
    if (totalQuestions == 0) return 0.0;
    return (correctAnswers / totalQuestions) * 100;
  }

  // Quiz devam ediyor mu
  bool get isRunning => status == QuizStatus.running;

  // Quiz tamamlandı mı
  bool get isCompleted => status == QuizStatus.completed;

  // Ortalama süre (soru başına saniye)
  int get averageTimePerQuestion {
    if (totalQuestions == 0) return 0;
    return (durationSeconds / totalQuestions).round();
  }

  factory QuizAttemptModel.fromJson(Map<String, dynamic> json) {
    return QuizAttemptModel(
      id: json['id'] as String? ?? '',
      userId: json['user_id'] as String? ?? '',
      subjectId: json['subject_id'] as String? ?? '',
      status: QuizStatus.fromString(json['status'] as String? ?? 'running'),
      totalQuestions: json['total_questions'] as int? ?? 0,
      correctAnswers: json['correct_answers'] as int? ?? 0,
      wrongAnswers: json['wrong_answers'] as int? ?? 0,
      blankAnswers: json['blank_answers'] as int? ?? 0,
      durationSeconds: json['duration_seconds'] as int? ?? 0,
      score: _parseDouble(json['score']),
      isTimed: json['is_timed'] as bool? ?? true,
      timeLimitSeconds: json['time_limit_seconds'] as int?,
      startedAt: DateTime.tryParse(json['started_at'] as String? ?? '') ??
          DateTime.now(),
      completedAt: json['completed_at'] != null
          ? DateTime.tryParse(json['completed_at'] as String)
          : null,
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'user_id': userId,
      'subject_id': subjectId,
      'status': status.name,
      'total_questions': totalQuestions,
      'correct_answers': correctAnswers,
      'wrong_answers': wrongAnswers,
      'blank_answers': blankAnswers,
      'duration_seconds': durationSeconds,
      'score': score,
      'is_timed': isTimed,
      'time_limit_seconds': timeLimitSeconds,
      'started_at': startedAt.toIso8601String(),
      'completed_at': completedAt?.toIso8601String(),
    };
  }

  QuizAttemptModel copyWith({
    String? id,
    String? userId,
    String? subjectId,
    QuizStatus? status,
    int? totalQuestions,
    int? correctAnswers,
    int? wrongAnswers,
    int? blankAnswers,
    int? durationSeconds,
    double? score,
    bool? isTimed,
    int? timeLimitSeconds,
    DateTime? startedAt,
    DateTime? completedAt,
  }) {
    return QuizAttemptModel(

@immutable
class StudyPlanWeekModel {
  final int weekNumber;
  final List<StudyPlanDayModel> days;
  final DateTime startDate;
  final DateTime endDate;

  const StudyPlanWeekModel({
    required this.weekNumber,
    required this.days,
    required this.startDate,
    required this.endDate,
  });

  factory StudyPlanWeekModel.fromJson(Map<String, dynamic> json) {
    final daysList = (json['days'] as List<dynamic>?)
            ?.map((day) => StudyPlanDayModel.fromJson(day as Map<String, dynamic>))
            .toList() ??
        [];

    return StudyPlanWeekModel(
      weekNumber: json['week_number'] as int? ?? 0,
      days: daysList,
      startDate: DateTime.tryParse(json['start_date'] as String? ?? '') ??
          DateTime.now(),
      endDate: DateTime.tryParse(json['end_date'] as String? ?? '') ??
          DateTime.now().add(const Duration(days: 7)),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'week_number': weekNumber,
      'days': days.map((day) => day.toJson()).toList(),
      'start_date': startDate.toIso8601String(),
      'end_date': endDate.toIso8601String(),
    };
  }

  StudyPlanWeekModel copyWith({
    int? weekNumber,
    List<StudyPlanDayModel>? days,
    DateTime? startDate,
    DateTime? endDate,
  }) {
    return StudyPlanWeekModel(
      weekNumber: weekNumber ?? this.weekNumber,
      days: days ?? this.days,
      startDate: startDate ?? this.startDate,
      endDate: endDate ?? this.endDate,
    );
  }
}

@immutable
class StudyPlanModel {
  final String id;
  final String userId;
  final List<StudyPlanWeekModel> weeks;
  final bool isActive;
  final DateTime createdAt;
  final DateTime updatedAt;

  const StudyPlanModel({
    required this.id,
    required this.userId,
    required this.weeks,
    required this.isActive,
    required this.createdAt,
    required this.updatedAt,
  });

  // Genel ilerleme yüzdesini hesapla
  double get overallProgress {
    int totalTasks = 0;
    int completedTasks = 0;

    for (final week in weeks) {
      for (final day in week.days) {
        totalTasks += day.tasks.length;
        completedTasks += day.tasks.where((task) => task.isCompleted).length;
      }
    }

    if (totalTasks == 0) return 0.0;
    return (completedTasks / totalTasks) * 100;
  }

  // Geçmiş haftaları al
  List<StudyPlanWeekModel> get pastWeeks {
    final now = DateTime.now();
    return weeks.where((week) => week.endDate.isBefore(now)).toList();
  }

  // Mevcut haftayı al
  StudyPlanWeekModel? get currentWeek {
    final now = DateTime.now();
    try {
      return weeks.firstWhere((week) =>
          week.startDate.isBefore(now.add(const Duration(days: 1))) &&
          week.endDate.isAfter(now));
    } catch (e) {
      return null;
    }
  }

  // Gelecek haftaları al
  List<StudyPlanWeekModel> get upcomingWeeks {
    final now = DateTime.now();
    return weeks.where((week) => week.startDate.isAfter(now)).toList();
  }

  factory StudyPlanModel.fromJson(Map<String, dynamic> json) {
    final weeksList = (json['plan_data'] as List<dynamic>?)
            ?.map((week) => StudyPlanWeekModel.fromJson(week as Map<String, dynamic>))
            .toList() ??
        [];

    return StudyPlanModel(
      id: json['id'] as String? ?? '',
      userId: json['user_id'] as String? ?? '',
      weeks: weeksList,
      isActive: json['is_active'] as bool? ?? true,
      createdAt: DateTime.tryParse(json['created_at'] as String? ?? '') ??
          DateTime.now(),
      updatedAt: DateTime.tryParse(json['updated_at'] as String? ?? '') ??

double? _parseDouble(dynamic value) {
  if (value == null) return null;
  if (value is double) return value;
  if (value is int) return value.toDouble();
  if (value is String) {
    return double.tryParse(value);
  }
  return null;
}
